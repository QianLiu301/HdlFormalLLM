                     line width=0.5pt, rounded corners=2pt},

  rowtitle/.style = {font=\scriptsize\bfseries, anchor=west, inner sep=0pt},

% ---- one row of the pipeline; #1 = node-name prefix, #2 = row title -------
\newcommand{\wfrow}[2]{%
  \node[rowtitle]              at (-0.68,0.72) {#2};
  \node[box, minimum width=13.5mm] (#1int) at (0,0)     {Design\\intent};
  \node[box]                       (#1duv) at (2.45,0.72)  {1.~DUV Gen};

  \node[note]                      at (5.30,-0.74) {deterministic,\\no LLM};
  % fan-out
  \draw[flow] (#1int.east) -- ++(0.35,0) |- (#1duv.west);
  \draw[flow] (#1int.east) -- ++(0.35,0) |- (#1bdd.west);
  % fan-in
  \draw[flow] (#1duv.east) -- ++(1.20,0) |- (#1tb.west);
  \draw[flow] (#1bdd.east) -- ++(1.20,0) |- (#1tb.west);
  \draw[flow] (#1tb.east)  -- (#1sim.west);
}
